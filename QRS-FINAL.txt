// ============================================================================
// 1. BASE DE TIEMPOS (Tick de 200 Hz)
// ============================================================================
module tick_200hz #(
    parameter DIV = 500000    // 100 MHz / 500000 = 200 Hz (en simulacion se reduce)
)(
    input clk,    // Reloj principal de 100 MHz
    input rst,    // Reset sincrono
    output tick   // Pulso de 1 ciclo cada 5ms (200 Hz)
);
    reg [18:0] count;
    always @(posedge clk) begin
        if (rst) begin
            count <= 0;
        end else if (count == DIV-1) begin
            count <= 0;
        end else begin
            count <= count + 1;
        end
    end
    assign tick = (count == DIV-1);
endmodule

// ============================================================================
// 2. MEMORIA ROM (Datos del ECG)
// ============================================================================
module rom_ecg (
    input clk,
    input [14:0] addr,       // 15 bits para abarcar 32768 posiciones
    output reg [11:0] data_out
);
    (* ram_style = "block" *) reg [11:0] memory [0:32767];

    initial begin
        $readmemh("rom_ecg_4seg.mem", memory);
    end

    always @(posedge clk) begin
        data_out <= memory[addr];
    end
endmodule

// ============================================================================
// 3. FILTRO PASA-BAJAS (LPF)
// ============================================================================
module lpf (
    input clk,
    input rst,
    input en,                     // Conectado al tick de 200 Hz (s0)
    input signed [11:0] x_in,
    output reg signed [17:0] y_out,
    output reg s1                 // Pulso para la siguiente etapa
);
    reg signed [11:0] x_delay [0:12];
    reg signed [17:0] y1, y2;
    integer i;

    wire signed [17:0] yn_next = (y1 <<< 1) - y2 + x_delay[0] - (x_delay[6] <<< 1) + x_delay[12];

    always @(posedge clk) begin
        if (rst) begin
            y_out <= 0; y1 <= 0; y2 <= 0; s1 <= 0;
            for (i = 0; i < 13; i = i + 1) x_delay[i] <= 0;
        end else begin
            s1 <= en;
            if (en) begin 
                for (i = 12; i > 0; i = i - 1) x_delay[i] <= x_delay[i-1];
                x_delay[0] <= x_in;
                y2 <= y1;
                y1 <= yn_next;
                y_out <= yn_next;
            end
        end
    end
endmodule

// ============================================================================
// 4. FILTRO PASA-ALTAS (HPF)
// ============================================================================
module hpf (
    input clk,
    input rst,
    input s1,
    input signed [17:0] pb_in,
    output reg signed [22:0] pa_out,
    output reg s2
);
    reg signed [17:0] x_delay [0:32];
    reg signed [22:0] p_prev;
    integer i;

    wire signed [22:0] p_next = p_prev + x_delay[0] - x_delay[32];
    wire signed [22:0] y_next = (x_delay[16] <<< 5) - p_next;

    always @(posedge clk) begin
        if (rst) begin
            pa_out <= 0; p_prev <= 0; s2 <= 0;
            for (i = 0; i < 33; i = i + 1) x_delay[i] <= 0;
        end else begin
            s2 <= s1;
            if (s1) begin
                for (i = 32; i > 0; i = i - 1) x_delay[i] <= x_delay[i-1];
                x_delay[0] <= pb_in;
                p_prev <= p_next;
                pa_out <= y_next;
            end
        end
    end
endmodule

// ============================================================================
// 5. DERIVADA Y SATURACION
// ============================================================================
module deriv (
    input clk,
    input rst,
    input s2,
    input signed [22:0] pa_in,
    output reg signed [15:0] der_out,
    output reg s3
);
    reg signed [22:0] x [0:4];
    integer i;

    wire signed [24:0] num = (x[0] <<< 1) + x[1] - x[3] - (x[4] <<< 1);
    wire signed [24:0] d_shift = num >>> 5;

    always @(posedge clk) begin
        if (rst) begin
            s3 <= 0; der_out <= 0;
            for (i = 0; i < 5; i = i + 1) x[i] <= 0;
        end else begin
            s3 <= s2; 
            if (s2) begin
                for (i = 4; i > 0; i = i - 1) x[i] <= x[i-1];
                x[0] <= pa_in;
                
                if (d_shift > 32767) der_out <= 32767;
                else if (d_shift < -32768) der_out <= -32768;
                else der_out <= d_shift[15:0];
            end
        end
    end
endmodule

// ============================================================================
// 6. CUADRADO (Usa DSP48E1)
// ============================================================================
module square (
    input clk,
    input rst,
    input s3,
    input signed [15:0] der_in,
    output reg signed [31:0] sq_out,
    output reg s4
);
    always @(posedge clk) begin
        if (rst) begin
            s4 <= 0; sq_out <= 0;
        end else begin
            s4 <= s3;
            if (s3) sq_out <= der_in * der_in;
        end
    end
endmodule

// ============================================================================
// 7. INTEGRACION (Ventana Movil)
// ============================================================================
module integ (
    input clk,
    input rst,
    input s4,
    input signed [31:0] sq_in,
    output reg signed [31:0] y_out,
    output reg s5
);
    reg signed [31:0] x [0:32];
    reg signed [36:0] S;
    integer i;

    always @(posedge clk) begin
        if (rst) begin
            s5 <= 0; y_out <= 0; S <= 0;
            for (i = 0; i < 33; i = i + 1) x[i] <= 0;
        end else begin
            s5 <= s4;
            if (s4) begin
                for (i = 32; i > 0; i = i - 1) x[i] <= x[i-1];
                x[0] <= sq_in;
                S <= S + sq_in - x[32];
                y_out <= (S + sq_in - x[32]) >>> 5; 
            end
        end
    end
endmodule

// ============================================================================
// 8. MAQUINA DE ESTADOS (Detector QRS)
// ============================================================================
module qrs_fsm (
    input clk,
    input rst,
    input s5,
    input signed [31:0] y_in,
    output reg beat,
    output reg s6
);
    localparam APRENDER = 3'd0, INICIAR = 3'd1, BUSCAR = 3'd2, 
               CLASIFICAR = 3'd3, LATIDO = 3'd4, RUIDO = 3'd5, REFRACTARIO = 3'd6;
               
    reg [2:0] state;
    reg signed [31:0] y_prev, d_prev, PEAKI, lmax, lsum, SPKI, NPKI;
    reg [8:0] n;
    reg [5:0] cnt;
    
    wire signed [32:0] d = y_in - y_prev;
    wire pk = (d_prev > 0 && d <= 0);
    wire signed [31:0] THR1 = NPKI + ((SPKI - NPKI) >>> 2);

    always @(posedge clk) begin
        if (rst) begin
            state <= APRENDER;
            beat <= 0; s6 <= 0; n <= 0; lmax <= 0; lsum <= 0;
            SPKI <= 0; NPKI <= 0; y_prev <= 0; d_prev <= 0; PEAKI <= 0; cnt <= 0;
        end else begin
            s6 <= s5; 
            if (s5) begin
                y_prev <= y_in; d_prev <= d; PEAKI <= y_prev; 
            end
            
            beat <= 0;
            
            case (state)
                APRENDER: begin
                    if (s5) begin
                        if (y_in > lmax) lmax <= y_in;
                        lsum <= lsum + y_in;
                        n <= n + 1;
                        if (n == 511) state <= INICIAR;
                    end
                end
                INICIAR: begin
                    SPKI <= lmax; NPKI <= lsum >>> 10; state <= BUSCAR;
                end
                BUSCAR: begin
                    if (s5 && pk) state <= CLASIFICAR;
                end
                CLASIFICAR: begin
                    if (PEAKI > THR1) state <= LATIDO;
                    else state <= RUIDO;
                end
                LATIDO: begin
                    beat <= 1; 
                    SPKI <= SPKI + ((PEAKI - SPKI) >>> 3);
                    cnt <= 0; state <= REFRACTARIO;
                end
                RUIDO: begin
                    NPKI <= NPKI + ((PEAKI - NPKI) >>> 3);
                    state <= BUSCAR;
                end
                REFRACTARIO: begin
                    if (s5) begin
                        cnt <= cnt + 1;
                        if (cnt == 39) state <= BUSCAR;
                    end
                end
            endcase
        end
    end
endmodule

// ============================================================================
// 10. CONTADOR DE LATIDOS EN BCD (4 digitos: 0000 a 9999)
// ============================================================================
module bcd_counter (
    input clk,
    input rst,
    input inc,                       // pulso 'beat' de 1 ciclo
    output reg [3:0] d0, d1, d2, d3  // d0 = unidades ... d3 = millares
);
    always @(posedge clk) begin
        if (rst) begin
            d0 <= 0; d1 <= 0; d2 <= 0; d3 <= 0;
        end else if (inc) begin
            if (d0 == 9) begin
                d0 <= 0;
                if (d1 == 9) begin
                    d1 <= 0;
                    if (d2 == 9) begin
                        d2 <= 0;
                        if (d3 == 9) d3 <= 0;
                        else d3 <= d3 + 1;
                    end else d2 <= d2 + 1;
                end else d1 <= d1 + 1;
            end else d0 <= d0 + 1;
        end
    end
endmodule

// ============================================================================
// 11. CALCULO DE BPM MEDIO DEL SEGMENTO
//     BPM = 12000 * (N-1) / (muestra_ultimo_latido - muestra_primer_latido)
//     (200 muestras/s * 60 s = 12000). Se calcula una sola vez al terminar
//     el segmento, con un divisor secuencial de 24 ciclos.
// ============================================================================
module bpm_calc (
    input clk,
    input rst,
    input beat,            // pulso de latido
    input [13:0] idx,      // contador de muestras
    input go,              // 1 = el segmento termino, calcular
    output reg [9:0] bpm,  // BPM medio (0 si hay menos de 2 latidos)
    output reg valid       // 1 = bpm ya esta calculado
);
    reg [9:0]  n;                  // latidos contados
    reg [13:0] first, last;        // muestra del primer y del ultimo latido
    reg        have_first;
    reg [1:0]  st;                 // 0 = espera, 1 = dividiendo, 2 = listo
    reg [23:0] num, quo;
    reg [24:0] rem;
    reg [13:0] den;
    reg [4:0]  cnt;

    wire [24:0] rem_sh = {rem[23:0], num[23]};
    wire        ge     = (rem_sh >= {11'd0, den});
    wire [23:0] quo_nx = {quo[22:0], ge};

    always @(posedge clk) begin
        if (rst) begin
            n <= 0; first <= 0; last <= 0; have_first <= 0;
            st <= 0; num <= 0; quo <= 0; rem <= 0; den <= 0; cnt <= 0;
            bpm <= 0; valid <= 0;
        end else begin
            if (beat && st == 0) begin
                n <= n + 1;
                if (!have_first) begin first <= idx; have_first <= 1; end
                last <= idx;
            end

            case (st)
                0: if (go) begin
                       if (n < 2 || last == first) begin
                           bpm <= 0; valid <= 1; st <= 2;
                       end else begin
                           num <= (n - 1) * 12000;
                           den <= last - first;
                           rem <= 0; quo <= 0; cnt <= 0; st <= 1;
                       end
                   end
                1: begin
                       num <= {num[22:0], 1'b0};
                       rem <= ge ? (rem_sh - {11'd0, den}) : rem_sh;
                       quo <= quo_nx;
                       cnt <= cnt + 1;
                       if (cnt == 23) begin
                           bpm   <= (quo_nx > 999) ? 10'd999 : quo_nx[9:0];
                           valid <= 1;
                           st    <= 2;
                       end
                   end
                default: ;
            endcase
        end
    end
endmodule

// ============================================================================
// 12. DISPLAY DE 7 SEGMENTOS (Basys 3: segmentos y anodos activos en 0)
//     Codigos de cada digito: 0-9 = numero, 10 = apagado, 11 = letra 'b'
// ============================================================================
module display_7seg (
    input clk,
    input [3:0] d0, d1, d2, d3,   // d0 = digito de la derecha
    output reg [6:0] seg,         // {g,f,e,d,c,b,a}
    output reg [3:0] an           // an[0] = digito de la derecha
);
    // Refresco: cada digito se enciende ~381 Hz (100 MHz / 2^18 * 4 digitos)
    reg [17:0] refresh = 0;
    always @(posedge clk) refresh <= refresh + 1;
    wire [1:0] sel = refresh[17:16];

    reg [3:0] digit;

    always @(*) begin
        case (sel)
            2'd0:    begin digit = d0; an = 4'b1110; end
            2'd1:    begin digit = d1; an = 4'b1101; end
            2'd2:    begin digit = d2; an = 4'b1011; end
            default: begin digit = d3; an = 4'b0111; end
        endcase

        case (digit)
            4'd0:  seg = 7'b1000000;
            4'd1:  seg = 7'b1111001;
            4'd2:  seg = 7'b0100100;
            4'd3:  seg = 7'b0110000;
            4'd4:  seg = 7'b0011001;
            4'd5:  seg = 7'b0010010;
            4'd6:  seg = 7'b0000010;
            4'd7:  seg = 7'b1111000;
            4'd8:  seg = 7'b0000000;
            4'd9:  seg = 7'b0010000;
            4'd11: seg = 7'b0000011;   // letra 'b'
            default: seg = 7'b1111111; // apagado
        endcase
    end
endmodule

// ============================================================================
// 9. TOP MODULE (Integracion Estructural)
// ============================================================================
module top_qrs #(
    parameter TICK_DIV = 500000,      // Ciclos de clk por muestra (500000 = 200 Hz a 100 MHz)
    parameter LED_HOLD = 10_000_000,  // Ciclos que dura el LED encendido (100 ms a 100 MHz)
    parameter BPM_DELAY = 300_000_000 // Pausa (3 s) mostrando el conteo final antes de mostrar el BPM
)(
    input clk,              
    input rst,              
    input [1:0] sw,         
    output beat_led,        
    output [6:0] seg,       
    output [3:0] an         
);

    wire s0_raw, s1, s2, s3, s4, s5, s6;
    reg  done;                 // 1 = ya se leyo todo el segmento elegido
    wire s0 = s0_raw & ~done;  // tick de muestreo, congelado al terminar el segmento
    wire [11:0] x_in;
    wire signed [17:0] pb_out;
    wire signed [22:0] pa_out;
    wire signed [15:0] der_out;
    wire signed [31:0] sq_out;
    wire signed [31:0] y_out;
    wire beat;
    
    reg [14:0] addr;
    
    always @(posedge clk) begin
        if (rst) begin
            addr <= {sw, 13'd0};   // sw elige el segmento (solo se lee durante el reset)
            done <= 1'b0;
        end else if (s0) begin
            addr <= addr + 1;
            if (addr[12:0] == 13'h1FFF) done <= 1'b1;   // ultima muestra del segmento
        end
    end

    tick_200hz #(.DIV(TICK_DIV)) u_tick (.clk(clk), .rst(rst), .tick(s0_raw));
    rom_ecg    u_rom  (.clk(clk), .addr(addr), .data_out(x_in));

    lpf u_lpf (
        .clk(clk), .rst(rst), .en(s0), 
        .x_in(x_in), .y_out(pb_out), .s1(s1)
    );
    
    hpf u_hpf (
        .clk(clk), .rst(rst), .s1(s1), 
        .pb_in(pb_out), .pa_out(pa_out), .s2(s2)
    );
    
    deriv u_deriv (
        .clk(clk), .rst(rst), .s2(s2), 
        .pa_in(pa_out), .der_out(der_out), .s3(s3)
    );
    
    square u_square (
        .clk(clk), .rst(rst), .s3(s3), 
        .der_in(der_out), .sq_out(sq_out), .s4(s4)
    );
    
    integ u_integ (
        .clk(clk), .rst(rst), .s4(s4), 
        .sq_in(sq_out), .y_out(y_out), .s5(s5)
    );
    
    qrs_fsm u_detector (
        .clk(clk), .rst(rst), .s5(s5), 
        .y_in(y_out), .beat(beat), .s6(s6)
    );

    // Logica para visualizar el latido en el LED[0] (100 ms)
    reg [23:0] led_timer;
    always @(posedge clk) begin
        if (rst) led_timer <= 0;
        else if (beat) led_timer <= LED_HOLD; 
        else if (led_timer > 0) led_timer <= led_timer - 1;
    end
    assign beat_led = (led_timer > 0);

    // ------------------------------------------------------------------
    // Contador de latidos (BCD)
    // ------------------------------------------------------------------
    wire [3:0] c0, c1, c2, c3;
    bcd_counter u_cnt (.clk(clk), .rst(rst), .inc(beat),
                       .d0(c0), .d1(c1), .d2(c2), .d3(c3));

    // ------------------------------------------------------------------
    // BPM medio del segmento (se calcula al terminar de leer la ROM)
    // ------------------------------------------------------------------
    reg [13:0] smp;                      // contador de muestras desde el reset
    always @(posedge clk) begin
        if (rst) smp <= 0;
        else if (s0) smp <= smp + 1;
    end

    reg [7:0] fin_cnt;                   // espera unos ciclos tras 'done' para
    wire fin = (fin_cnt == 8'hFF);       // que salgan los ultimos latidos
    always @(posedge clk) begin
        if (rst) fin_cnt <= 0;
        else if (done && !fin) fin_cnt <= fin_cnt + 1;
    end

    wire [9:0] bpm;
    wire       bpm_valid;
    bpm_calc u_bpm (.clk(clk), .rst(rst), .beat(beat), .idx(smp), .go(fin),
                    .bpm(bpm), .valid(bpm_valid));

    // Pausa para alcanzar a ver el conteo final antes de pasar al BPM
    reg [28:0] dly;
    wire show_bpm = (dly == BPM_DELAY);
    always @(posedge clk) begin
        if (rst) dly <= 0;
        else if (bpm_valid && !show_bpm) dly <= dly + 1;
    end

    // ------------------------------------------------------------------
    // Que se muestra: latidos mientras corre, 'b' + BPM al final
    // ------------------------------------------------------------------
    localparam BL = 4'd10;   // digito apagado
    localparam LB = 4'd11;   // letra 'b'

    wire [3:0] bh = bpm / 100;
    wire [3:0] bt = (bpm % 100) / 10;
    wire [3:0] bu = bpm % 10;

    reg [3:0] e0, e1, e2, e3;
    always @(*) begin
        if (show_bpm) begin
            e0 = bu;
            e1 = (bh == 0 && bt == 0) ? BL : bt;
            e2 = (bh == 0) ? BL : bh;
            e3 = LB;
        end else begin
            e0 = c0;
            e1 = (c3 == 0 && c2 == 0 && c1 == 0) ? BL : c1;
            e2 = (c3 == 0 && c2 == 0) ? BL : c2;
            e3 = (c3 == 0) ? BL : c3;
        end
    end

    display_7seg u_disp (.clk(clk), .d0(e0), .d1(e1), .d2(e2), .d3(e3),
                         .seg(seg), .an(an));

endmodule
